    # ---- fixed acceptance pairs, review 2026-10-17 §4.3 (signed charge account on the vm ledger)
    from netlists.ledger_plan import membrane_ledger
    from observers.support_q1r import consts as _consts, score_events

    _cert = _proto_cert()[1]

    def cdir_record(vm_traj, i_in, kh=None, q_times=(), hc=None, close=True):
        """Synthetic Cdir record on the real CHAIN_12_W_F ledger plan: written vm trajectory,
        injector current into vm, kh shunt OUT of vm (default zero but measured), optional hc."""
        V = _cp(L0)
        V[vmk] = np.asarray(vm_traj, dtype=float)
        V[mo["vec"].lower()] = -np.asarray(i_in, dtype=float) / mo["sign"]
        V[kk["vec"].lower()] = (np.zeros_like(t) if kh is None else np.asarray(kh, dtype=float)) / kk["sign"]
        V["v(cdw12f_hs)"] = np.full_like(t, 1.3)
        if hc is not None:
            V["v(cdw12f_hc)"] = np.asarray(hc, dtype=float)
        V["v(cdw12f_q)"] = np.zeros_like(t)
        V["v(cdw12f_op_aq)"] = np.zeros_like(t)
        V["v(cdw12f_op_p)"] = np.zeros_like(t)
        for tq in q_times:
            V["v(cdw12f_q)"] += np.where((t >= tq) & (t < tq + 0.02), 2.967, 0.0)
        if close:
            _close_ledger(V, plan_, t)      # real reset / discharge branch carries the remainder
        return V, membrane_ledger(V, plan_, "cdw12f_")

    def cdir_events(led, cause):

        return score_events(V_cur[0], "cdw12f_", cause & (V_cur[0]["v(cdw12f_hs)"] >= K["H_GATE"]),

    def rc_traj():
        """Two supported stretches (ramp to entry, hold through q, passive decay) with the
        injector supplying the written node equation (Cm dv/dt + leak, 5 % margin)."""
        Cmm, Rmm, tau = 1.5e-6, 1e6, 1.5
        tq1, tq2, ramp = 1.4, 3.4, 0.6
        vm = np.full_like(t, vrest0)
        i_in = np.zeros_like(t)
        for tq in (tq1, tq2):
            a = tq - ramp
            v0 = float(np.interp(a, t, vm))
            r = (t >= a) & (t < tq)
            vm[r] = v0 + (vrest0 + ue_tc - v0) * (t[r] - a) / ramp
            vm[(t >= tq) & (t < tq + 0.02)] = vrest0 + ue_tc
            dec = t >= tq + 0.02
            vm[dec] = vrest0 + ue_tc * np.exp(-(t[dec] - (tq + 0.02)) / tau)
            i_in = i_in + np.where(r, 1.05 * Cmm * np.gradient(vm, t) + (vm - vrest0) / Rmm, 0.0)
        return vm, i_in, [(tq1 - ramp, tq1 + 0.1), (tq2 - ramp, tq2 + 0.1)], (tq1, tq2)
